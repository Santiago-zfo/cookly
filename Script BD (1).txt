-- 5. PROCEDIMIENTOS ALMACENADOS (Consultas del sistema)

DELIMITER //

CREATE PROCEDURE sp_consultar_recetas_detalladas()
BEGIN
    SELECT r.id_receta, r.titulo, r.descripcion, r.instrucciones, r.tiempo_preparacion, 
           r.dificultad, r.porciones, r.estado, r.imagen_url, c.nombre AS categoria, 
           u.nombre AS autor, u.correo_electronico AS correo_autor
    FROM Recetas r
    INNER JOIN Categorias c ON r.id_categoria = c.id_categoria
    INNER JOIN Usuarios u ON r.id_usuario = u.id_usuario;
END //

CREATE PROCEDURE sp_consultar_ingredientes_receta()
BEGIN
    SELECT ri.id_receta, r.titulo AS receta, i.nombre AS ingrediente, ri.cantidad, ri.unidad_medida
    FROM Receta_Ingredientes ri
    INNER JOIN Recetas r ON ri.id_receta = r.id_receta
    INNER JOIN Ingredientes i ON ri.id_ingrediente = i.id_ingrediente
    ORDER BY ri.id_receta, i.nombre;
END //

CREATE PROCEDURE sp_consultar_recetas_por_categoria()
BEGIN
    SELECT c.id_categoria, c.nombre AS categoria, COUNT(r.id_receta) AS total_recetas
    FROM Categorias c
    LEFT JOIN Recetas r ON c.id_categoria = r.id_categoria
    GROUP BY c.id_categoria, c.nombre;
END //

CREATE PROCEDURE sp_consultar_recetas_por_usuario()
BEGIN
    SELECT u.id_usuario, u.nombre AS usuario, u.correo_electronico, COUNT(r.id_receta) AS total_recetas
    FROM Usuarios u
    LEFT JOIN Recetas r ON u.id_usuario = r.id_usuario
    GROUP BY u.id_usuario, u.nombre, u.correo_electronico;
END //

CREATE PROCEDURE sp_consultar_receta_completa(IN p_id_receta INT)
BEGIN
    SELECT r.titulo AS receta, c.nombre AS categoria, u.nombre AS autor, i.nombre AS ingrediente, ri.cantidad, ri.unidad_medida
    FROM Recetas r
    INNER JOIN Categorias c ON r.id_categoria = c.id_categoria
    INNER JOIN Usuarios u ON r.id_usuario = u.id_usuario
    INNER JOIN Receta_Ingredientes ri ON r.id_receta = ri.id_receta
    INNER JOIN Ingredientes i ON ri.id_ingrediente = i.id_ingrediente
    WHERE r.id_receta = p_id_receta
    ORDER BY i.nombre;
END //

DELIMITER ;
